=== run_campaign.sh ===
#!/bin/bash
set -uo pipefail
source /data/esierra/miniconda3/etc/profile.d/conda.sh
conda activate boltzgen
BASE=/data/esierra/proyecto3-boltzgen

echo "=== RBD campaign start $(date) ==="
cd "$BASE/rbd_campaign"
boltzgen run --protocol nanobody-anything --output run1 --devices 4 --num_designs 10000 --diffusion_batch_size 1 --reuse --budget 100 design_spec.yaml
rc=$?
echo "=== RBD campaign done rc=$rc $(date) ==="

echo "=== PfCSP campaign start $(date) ==="
cd "$BASE/pfcsp_campaign"
boltzgen run --protocol nanobody-anything --output run1 --devices 4 --num_designs 10000 --diffusion_batch_size 1 --reuse --budget 100 design_spec.yaml
rc2=$?
echo "=== PfCSP campaign done rc=$rc2 $(date) ==="
